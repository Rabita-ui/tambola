/* AUTOMATICALLY HEAL OR CREATE MISSING FIRESTORE USER DOC ON LOGIN */ async function loginUser(email, password) { try { const userCredential = await auth.signInWithEmailAndPassword(email, password); const user = userCredential.user; const userRef = db.collection("users").doc(user.uid); const userDoc = await userRef.get(); // If user doc doesn't exist in Firestore, create it now! if (!userDoc.exists) { await userRef.set({ uid: user.uid, email: user.email, displayName: user.email ? user.email.split("@")[0] : "Player", tokens: 0, role: "player", createdAt: firebase.firestore.FieldValue.serverTimestamp() }); } window.location.href = "dashboard.html"; } catch (error) { console.error("Login Error:", error); alert("Login failed: " + error.message); } }
