/* CREATE BOTH AUTH USER AND FIRESTORE USER DOC */ async function registerUser(email, password) { try { // 1. Create Auth account const userCredential = await auth.createUserWithEmailAndPassword(email, password); const user = userCredential.user; // 2. Immediately create user document in Firestore await db.collection("users").doc(user.uid).set({ uid: user.uid, email: user.email, displayName: email.split("@")[0], tokens: 0, role: "player", createdAt: firebase.firestore.FieldValue.serverTimestamp() }); alert("Account created successfully!"); window.location.href = "dashboard.html"; } catch (error) { console.error("Registration Error:", error); alert("Registration failed: " + error.message); } }
