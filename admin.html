// Add this inside admin.html script section async function seed600MasterTickets() { const confirmation = confirm("This will generate and upload 600 Master Tickets to Firestore. Continue?"); if (!confirmation) return; console.log("Generating 600 master tickets..."); const masterRef = db.collection("masterTickets"); // Process in two batches of 300 to stay under Firestore's 500 batch limit for (let batchNum = 0; batchNum < 2; batchNum++) { const batch = db.batch(); for (let i = 1; i <= 300; i++) { const ticketIndex = batchNum * 300 + i; const ticketId = `ticket_${String(ticketIndex).padStart(3, '0')}`; const ticketGrid = generateTambolaTicket(); // 3x9 grid generator algorithm const docRef = masterRef.doc(ticketId); batch.set(docRef, { ticketNumber: ticketIndex, grid: ticketGrid, createdAt: firebase.firestore.FieldValue.serverTimestamp() }); } await batch.commit(); console.log(`Batch ${batchNum + 1}/2 committed (300 tickets).`); } alert("Success! 600 Master Tickets are permanently stored in Firestore."); } // Standard 3x9 Tambola Generator Algorithm function generateTambolaTicket() { let ticket = Array.from({ length: 3 }, () => Array(9).fill(0)); let colRanges = [[1, 9], [10, 19], [20, 29], [30, 39], [40, 49], [50, 59], [60, 69], [70, 79], [80, 90]]; for (let r = 0; r < 3; r++) { let cols = []; while (cols.length < 5) { let col = Math.floor(Math.random() * 9); if (!cols.includes(col)) cols.push(col); } cols.sort((a, b) => a - b); for (let col of cols) { let [min, max] = colRanges[col]; let num; do { num = Math.floor(Math.random() * (max - min + 1)) + min; } while (ticket.some(row => row[col] === num)); ticket[r][col] = num; } } for (let c = 0; c < 9; c++) { let colNums = []; for (let r = 0; r < 3; r++) { if (ticket[r][c] !== 0) colNums.push(ticket[r][c]); } colNums.sort((a, b) => a - b); let idx = 0; for (let r = 0; r < 3; r++) { if (ticket[r][c] !== 0) ticket[r][c] = colNums[idx++]; } } return ticket; }
