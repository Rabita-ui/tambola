document.getElementById('create-game-form').addEventListener('submit', async (e) => { e.preventDefault(); const title = document.getElementById('game-title').value; const ticketPrice = Number(document.getElementById('ticket-price').value); const startTime = document.getElementById('game-time').value; const statusText = document.getElementById('game-status'); try { statusText.style.color = '#006644'; statusText.innerText = 'Creating game and generating tickets...'; // 1. Create Game Document const gameRef = await db.collection('games').add({ title: title, ticketPrice: ticketPrice, startTime: new Date(startTime), status: 'upcoming', drawnNumbers: [], createdAt: firebase.firestore.FieldValue.serverTimestamp() }); // 2. Generate Master Pool of 30 Default Tickets for this Game const batch = db.batch(); for (let i = 1; i <= 30; i++) { const ticketRef = gameRef.collection('tickets').doc(`ticket_${i}`); batch.set(ticketRef, { ticketNumber: i, matrix: generateTambolaTicket(), isBooked: false, bookedBy: null }); } await batch.commit(); statusText.innerText = 'Game and 30 master tickets created successfully!'; document.getElementById('create-game-form').reset(); } catch (error) { statusText.style.color = '#ff5630'; statusText.innerText = 'Failed to create game: ' + error.message; } }); // Helper: Standard 3x9 Matrix Generator function generateTambolaTicket() { let grid = Array(3).fill(null).map(() => Array(9).fill(0)); for (let col = 0; col < 9; col++) { let min = col === 0 ? 1 : col * 10; let max = col === 8 ? 90 : col * 10 + 9; let nums = []; while (nums.length < 2) { let rand = Math.floor(Math.random() * (max - min + 1)) + min; if (!nums.includes(rand)) nums.push(rand); } nums.sort((a, b) => a - b); grid[0][col] = nums[0]; grid[1][col] = nums[1]; } return grid; }
